# Dependencies for the RunPod A100 pipeline.
#
# Used by scripts/runpod/run_card_pipeline.sh on top of the
# `runpod/pytorch:0.7.0-cu1281-torch271-ubuntu2204` base image, which
# ships with Python 3.13 + torch 2.7.1 + CUDA 12.8.1.
#
# vLLM 0.10.2 is the oldest release that supports Python 3.13. It pins
# torch==2.8.0 (one minor patch ahead of the image's 2.7.1) — pip will
# upgrade torch transparently. CUDA 12.8 supports torch 2.8 fine.
#
# The main requirements.txt stays pinned to vllm 0.5.4 / torch 2.3 for
# local Mac/mlx development.

# vLLM compatible with Python 3.13
vllm==0.10.2

# transformers compatible with vllm 0.10.2.
# Upper-bound at <5.0 because transformers 5.x removed
# `tokenizer.all_special_tokens_extended`, which vllm 0.10.2 still
# calls when loading CodeLlama's slow tokenizer — fails at LLM(...)
# init time with AttributeError.
transformers>=4.55.2,<5.0
accelerate

# Numerics — leave unpinned so pip can match transformers' numpy>=2
# expectations.
numpy
scipy
scikit-learn

# Data
datasets
huggingface-hub

# Static analyzer parser (API is version-sensitive, pin tightly)
tree-sitter==0.23.2
tree-sitter-python==0.23.6

# Retrieval
rank-bm25

# Metrics
python-Levenshtein

# CARD's Estimator
lightgbm
xgboost

# Utilities
tqdm
pydantic
jsonlines
click
